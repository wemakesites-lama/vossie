# CampusCart logo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Eduvos crest with the CampusCart cart mark in the header pennant, the loading screen, the app icons and the favicon, as the logo spec says (`docs/superpowers/specs/2026-11-04-campuscart-logo-design.md`).

**Architecture:** One geometry module (`lib/brand/cart-mark.ts`) is the only drawing of the mark. A Server Component (`CartMark`) draws it inline in pages, coloured by theme tokens. A script (`scripts/make-icons.ts`, run by hand) draws the icon files and the master SVGs from the same module through sharp. The header pennant moves out of the client `SiteHeader` into a slot filled by the server layout, so the mark adds no JavaScript.

**Tech Stack:** Next.js 16.4 (App Router, Server Components), React 19.3, Tailwind CSS 4.3 component CSS in `app/globals.css`, Vitest 5 + Testing Library, Playwright, Node 26 type stripping for the script, sharp 0.35.5 (resolved through Next's own dependencies, not a new project dependency).

- [ ] **Step 3: Write the module**

`lib/brand/cart-mark.ts`:

```ts
/**
 * The CampusCart mark (logo spec, docs/superpowers/specs/2026-11-04-campuscart-logo-design.md): a line-drawn cart whose
 * rim is the Eduvos crest's top edge (the handle's grip and the raised right end are its corners, with its dip in the
 * middle), holding the crest's three stars, on wheels that echo the crest's ring.
 *
 * The only drawing of the mark. CartMark (components/brand/CartMark.tsx) draws it in pages; scripts/make-icons.ts draws
 * the app icons, the favicon and the masters in assets/brand from it. It imports nothing, so node can run the script.
 */

export type MarkCut = "full" | "small";

/** A four-point star: its centre and half its height. */
export type Star = readonly [x: number, y: number, size: number];

type Drawing = {
  handle: string;
  rim: string;
  /** Stroke width of the handle and the rim. */
  line: number;
  wheels: readonly (readonly [x: number, y: number])[];
  wheelRadius: number;
  wheelLine: number;
  smallStars: readonly Star[];
  bigStar: Star;
  /** What the strokes cover, round caps included: the icons centre this, not the grid. */
  box: { x: number; y: number; width: number; height: number };
};

export const MARK_WIDTH = 200;
export const MARK_HEIGHT = 175;
export const MARK_VIEWBOX = `0 0 ${MARK_WIDTH} ${MARK_HEIGHT}`;

/** The brand's hexes, for generated files only (icons, masters). Pages colour the mark with tokens (CartMark). */
export const NAVY = "#15305d";
export const SKY = "#29abe2";
export const WHITE = "#ffffff";

const HANDLE = "M10 24H32L58 124H174";
const WHEELS = [
  [84, 150],
  [156, 150],
] as const;

export const MARK: Record<MarkCut, Drawing> = {
  // 48 px and larger (spec 2.1).
  full: {
    handle: HANDLE,
    rim: "M38 46H104Q117 46 117 56Q117 46 130 46H170Q182 46 182 32H192L177 106H54",
    line: 11,
    wheels: WHEELS,
    wheelRadius: 11,
    wheelLine: 10,
    smallStars: [
      [88, 79, 11],
      [148, 79, 11],
    ],
    bigStar: [118, 81, 19],
    box: { x: 4, y: 18, width: 194, height: 149 },
  },
  // 32 px and smaller: heavier lines, a deeper rim, one star (spec 2.1).
  small: {
    handle: HANDLE,
    rim: "M38 46H101Q117 46 117 60Q117 46 133 46H166Q182 46 182 28H192L177 106H54",
    line: 17,
    wheels: WHEELS,
    wheelRadius: 11,
    wheelLine: 14,
    smallStars: [],
    bigStar: [118, 79, 20],
    box: { x: 1, y: 15, width: 200, height: 154 },
  },
};

export function starPath([x, y, s]: Star): string {
  return `M${x} ${y - s}Q${x} ${y} ${x + s} ${y}Q${x} ${y} ${x} ${y + s}Q${x} ${y} ${x - s} ${y}Q${x} ${y} ${x} ${y - s}Z`;
}

/** The mark's shapes as SVG markup, in the 200 × 175 grid: lines and small stars in `lines`, the big star in `star`. */
export function markShapes(cut: MarkCut, lines: string, star: string): string {
  const d = MARK[cut];
  const stroke = (path: string) =>
    `<path d="${path}" fill="none" stroke="${lines}" stroke-width="${d.line}" stroke-linecap="round" stroke-linejoin="round"/>`;
  const wheels = d.wheels.map(
    ([x, y]) => `<circle cx="${x}" cy="${y}" r="${d.wheelRadius}" fill="none" stroke="${lines}" stroke-width="${d.wheelLine}"/>`,
  );
  const small = d.smallStars.length > 0 ? `<path d="${d.smallStars.map(starPath).join("")}" fill="${lines}"/>` : "";
  return [stroke(d.handle), stroke(d.rim), ...wheels, small, `<path d="${starPath(d.bigStar)}" fill="${star}" class="cart-mark-star"/>`].join("");
}

/** A master SVG file of the mark (assets/brand/campuscart-mark-*.svg). */
export function markSvg(cut: MarkCut, lines: string, star: string): string {
  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="${MARK_VIEWBOX}" width="${MARK_WIDTH}" height="${MARK_HEIGHT}" overflow="visible">${markShapes(cut, lines, star)}</svg>\n`;
}
```

- [ ] **Step 3: Write the component**

`components/brand/CartMark.tsx`:

```tsx
import { MARK, MARK_VIEWBOX, starPath, type MarkCut } from "@/lib/brand/cart-mark";
import { cx } from "@/lib/cx";

type CartMarkProps = {
  /** "full" from 48 px up; "small" below (logo spec 2.1). */
  cut?: MarkCut;
  /** The big star in sky. Only on a band (the pennant): sky never sits on white (logo spec 2.3). */
  sky?: boolean;
  className?: string;
};

/**
 * The CampusCart mark, inline, in the text colour (`currentColor`), so each theme colours it with no image swap.
 * A Server Component: keep it out of client components (the header takes the pennant as a slot), so it costs no JS.
 * Decorative: the link or status around it carries the words. Size it with CSS.
 */
export function CartMark({ cut = "full", sky = false, className }: CartMarkProps) {
  const d = MARK[cut];
  return (
    <svg
      aria-hidden="true"
      focusable="false"
      viewBox={MARK_VIEWBOX}
      overflow="visible"
      className={cx("cart-mark", sky && "cart-mark--sky", className)}
    >
      <g fill="none" stroke="currentColor" strokeLinecap="round" strokeLinejoin="round">
        <path d={d.handle} strokeWidth={d.line} />
        <path d={d.rim} strokeWidth={d.line} />
        {d.wheels.map(([x, y]) => (
          <circle key={x} cx={x} cy={y} r={d.wheelRadius} strokeWidth={d.wheelLine} />
        ))}
      </g>
      {d.smallStars.length > 0 && <path d={d.smallStars.map(starPath).join("")} fill="currentColor" />}
      <path d={starPath(d.bigStar)} fill="currentColor" className="cart-mark-star" />
    </svg>
  );
}
```

`cx` (`lib/cx.ts`) skips `false`, so `sky && "cart-mark--sky"` is safe.

`lib/brand/ico.ts`:

```ts
/**
 * An .ico file of PNG images (app/favicon.ico, logo spec 3.3). Every current browser reads PNG entries. Each entry is
 * 16 bytes: width and height (0 means 256), no palette, 1 plane, 32 bits, the PNG's length and where it starts.
 */
export function icoFrom(images: readonly { size: number; png: Uint8Array }[]): Buffer {
  const header = Buffer.alloc(6);
  header.writeUInt16LE(0, 0);
  header.writeUInt16LE(1, 2);
  header.writeUInt16LE(images.length, 4);

  const directory = Buffer.alloc(16 * images.length);
  let offset = header.length + directory.length;
  images.forEach(({ size, png }, i) => {
    const at = i * 16;
    directory.writeUInt8(size >= 256 ? 0 : size, at);
    directory.writeUInt8(size >= 256 ? 0 : size, at + 1);
    directory.writeUInt8(0, at + 2);
    directory.writeUInt8(0, at + 3);
    directory.writeUInt16LE(1, at + 4);
    directory.writeUInt16LE(32, at + 6);
    directory.writeUInt32LE(png.length, at + 8);
    directory.writeUInt32LE(offset, at + 12);
    offset += png.length;
  });

  return Buffer.concat([header, directory, ...images.map(({ png }) => Buffer.from(png))]);
}
```

import { MARK, markShapes, NAVY, SKY, WHITE, type MarkCut } from "./cart-mark.ts";

/**
 * The app's icon files (logo spec 3.3): a navy square with the white CampusCart mark and its sky star. The names don't
 * change, so app/manifest.ts and Next's file conventions need nothing. scripts/make-icons.ts writes them.
 */
export type IconSpec = {
  size: number;
  cut: MarkCut;
  /** The mark's drawn width as a share of the icon's. */
  markWidth: number;
  /** Corner radius as a share of the size; 0 is square (Android and iOS round the corners themselves). */
  corner: number;
};

export const ICON_FILES: readonly (IconSpec & { file: string })[] = [
  { file: "public/icons/icon-192.png", size: 192, cut: "full", markWidth: 0.72, corner: 0 },
  { file: "public/icons/icon-512.png", size: 512, cut: "full", markWidth: 0.72, corner: 0 },

  { file: "app/apple-icon.png", size: 180, cut: "full", markWidth: 0.7, corner: 0 },
  // 64 × 64, shown by browsers at 32 or 16: the small cut.
  { file: "app/icon.png", size: 64, cut: "small", markWidth: 0.84, corner: 0.18 },
];

export const FAVICON = {
  file: "app/favicon.ico",
  corner: 0.18,
  images: [
    { size: 16, cut: "small", markWidth: 0.9 },
    { size: 32, cut: "small", markWidth: 0.88 },
    { size: 48, cut: "full", markWidth: 0.84 },
  ],
} as const satisfies { file: string; corner: number; images: readonly Omit<IconSpec, "corner">[] };

/** One icon as an SVG: the mark's drawn box centred in the navy square. */
export function iconSvg({ size, cut, markWidth, corner }: IconSpec): string {
  const { box } = MARK[cut];
  const scale = (size * markWidth) / box.width;
  const x = (size - box.width * scale) / 2 - box.x * scale;
  const y = (size - box.height * scale) / 2 - box.y * scale;
  return (
    `<svg xmlns="http://www.w3.org/2000/svg" width="${size}" height="${size}" viewBox="0 0 ${size} ${size}">` +
    `<rect width="${size}" height="${size}" rx="${size * corner}" fill="${NAVY}"/>` +
    `<g transform="translate(${x} ${y}) scale(${scale})">${markShapes(cut, WHITE, SKY)}</g></svg>`
  );
}

/**
 * Writes CampusCart's app icons, favicon and the mark's master SVGs from lib/brand (logo spec 3.3 and 4).
 * Run from the project root whenever the mark changes:  pnpm icons
 *
 * sharp isn't a project dependency: it comes with Next, so it's found through next's own package.json.

import { mkdirSync, writeFileSync } from "node:fs";
import { createRequire } from "node:module";
import { dirname, join } from "node:path";
import { markSvg, NAVY, SKY, WHITE } from "../lib/brand/cart-mark.ts";
import { icoFrom } from "../lib/brand/ico.ts";
import { FAVICON, ICON_FILES, iconSvg } from "../lib/brand/icons.ts";

type Sharp = (input: Buffer) => { png: (options: { compressionLevel: number }) => { toBuffer: () => Promise<Buffer> } };

const root = join(import.meta.dirname, "..");
const fromNext = createRequire(createRequire(import.meta.url).resolve("next/package.json"));
const sharp = fromNext("sharp") as Sharp;

const png = (svg: string) => sharp(Buffer.from(svg)).png({ compressionLevel: 9 }).toBuffer();

function write(file: string, data: string | Uint8Array) {
  const path = join(root, file);
  mkdirSync(dirname(path), { recursive: true });
  writeFileSync(path, data);
  console.log(`wrote ${file}`);
}

// Navy for white surfaces (all stars navy); white with the sky star for navy ones (logo spec 2.3).
const masters = [
  ["campuscart-mark-navy.svg", "full", NAVY, NAVY],
  ["campuscart-mark-white.svg", "full", WHITE, SKY],
  ["campuscart-mark-small-navy.svg", "small", NAVY, NAVY],
  ["campuscart-mark-small-white.svg", "small", WHITE, SKY],
] as const;
for (const [name, cut, lines, star] of masters) write(`assets/brand/${name}`, markSvg(cut, lines, star));

for (const icon of ICON_FILES) write(icon.file, await png(iconSvg(icon)));

const favicon = await Promise.all(
  FAVICON.images.map(async (image) => ({ size: image.size, png: await png(iconSvg({ ...image, corner: FAVICON.corner })) })),
);
write(FAVICON.file, icoFrom(favicon));
